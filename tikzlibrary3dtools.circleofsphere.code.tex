\ExplSyntaxOn
% \tdplotCsIfStyleT{<style list>}{<style name>}{<code>}
\NewDocumentCommand \tdplotCsIfStyleT { m m +m } { \tl_if_in:nnT {#1} {#2} {#3} }
% \tdplotCsIfTF{<flag>}{<true>}{<false>}: <flag> is a pgfmath comparison (1 or 0)
\NewDocumentCommand \tdplotCsIfTF { m +m +m } { \int_compare:nNnTF {#1} = { 1 } {#2} {#3} }
\msg_new:nnn { tikz-3dplot-circleofsphere } { elevation-out-of-range }
  {
    Elevation~angle~'#1'~is~not~in~the~open~interval~(-90,90).\\
    The~circle~would~degenerate~to~a~point,~so~nothing~is~drawn.
  }
\NewDocumentCommand \tdplotCsElevationError { m }
  { \msg_error:nnn { tikz-3dplot-circleofsphere } { elevation-out-of-range } {#1} }
\ExplSyntaxOff
% the full rotation matrix of tikz-3dplot (main and rotated coordinates):
% \axx ... \azz
\NewDocumentCommand{\tdplotCsComputeTransformRotScreen}{}{%
  \let\tdplotCsA\tdplotalpha
  \let\tdplotCsB\tdplotbeta
  \let\tdplotCsP\tdplotmainphi
  \let\tdplotCsT\tdplotmaintheta
  \pgfmathsetmacro\axx{cos(\tdplotCsA)*cos(\tdplotCsB)*cos(\tdplotCsP) + cos(\tdplotCsB)*sin(\tdplotCsA)*sin(\tdplotCsP)}
  \pgfmathsetmacro\axy{cos(\tdplotCsA)*sin(\tdplotCsP) - cos(\tdplotCsP)*sin(\tdplotCsA)}
  \pgfmathsetmacro\axz{cos(\tdplotCsA)*cos(\tdplotCsP)*sin(\tdplotCsB) + sin(\tdplotCsA)*sin(\tdplotCsB)*sin(\tdplotCsP)}
  \pgfmathsetmacro\ayx{cos(\tdplotCsB)*cos(\tdplotCsP)*sin(\tdplotCsA)*cos(\tdplotCsT) - cos(\tdplotCsA)*cos(\tdplotCsB)*cos(\tdplotCsT)*sin(\tdplotCsP) - sin(\tdplotCsB)*sin(\tdplotCsT)}
  \pgfmathsetmacro\ayy{cos(\tdplotCsA)*cos(\tdplotCsP)*cos(\tdplotCsT) + sin(\tdplotCsA)*cos(\tdplotCsT)*sin(\tdplotCsP)}
  \pgfmathsetmacro\ayz{cos(\tdplotCsB)*sin(\tdplotCsT) - cos(\tdplotCsA)*sin(\tdplotCsB)*cos(\tdplotCsT)*sin(\tdplotCsP) + cos(\tdplotCsP)*sin(\tdplotCsA)*sin(\tdplotCsB)*cos(\tdplotCsT)}
  \pgfmathsetmacro\azx{cos(\tdplotCsA)*cos(\tdplotCsB)*sin(\tdplotCsP)*sin(\tdplotCsT) - sin(\tdplotCsB)*cos(\tdplotCsT) - cos(\tdplotCsB)*cos(\tdplotCsP)*sin(\tdplotCsA)*sin(\tdplotCsT)}
  \pgfmathsetmacro\azy{-cos(\tdplotCsA)*cos(\tdplotCsP)*sin(\tdplotCsT) - sin(\tdplotCsA)*sin(\tdplotCsP)*sin(\tdplotCsT)}
  \pgfmathsetmacro\azz{cos(\tdplotCsB)*cos(\tdplotCsT) + cos(\tdplotCsA)*sin(\tdplotCsB)*sin(\tdplotCsP)*sin(\tdplotCsT) - cos(\tdplotCsP)*sin(\tdplotCsA)*sin(\tdplotCsB)*sin(\tdplotCsT)}
}
% \tdplotCsDrawCircle[<style>]{<sphere radius>}{<alpha>}{<beta>}{<elevation>}:
% the circle whose plane has the normal (alpha, beta) (azimuth, polar angle)
% and lies at the elevation angle above the center; the style list may set
% tdplotCsFront, tdplotCsBack and, to fill the circle, tdplotCsFill
\NewDocumentCommand{\tdplotCsDrawCircle}{O{}mmmm}{%
  \pgfmathsetmacro\tdplotCsEps{#5}%
  \tdplotCsIfTF{\fpeval{abs(\tdplotCsEps)>=90}}{\tdplotCsElevationError{#5}}{%
    \begin{scope}[#1]
      \tdplotCsIfStyleT{#1}{tdplotCsDrawAux}{%
        \PackageWarning{3dtools}{tdplotCsDrawAux is not supported}}%
      \pgfmathsetmacro\tdplotCsAlp{#3}%
      \pgfmathsetmacro\tdplotCsBet{#4}%
      \edef\tdplotCsN{\fpeval{cosd(\tdplotCsAlp)*sind(\tdplotCsBet)},%
        \fpeval{sind(\tdplotCsAlp)*sind(\tdplotCsBet)},\fpeval{cosd(\tdplotCsBet)}}%
      \pgfmathsetmacro\tdplotCsZe{(#2)*sin(\tdplotCsEps)}%
      \edef\tdplotCsP{\fpeval{\tdplotCsZe*cosd(\tdplotCsAlp)*sind(\tdplotCsBet)},%
        \fpeval{\tdplotCsZe*sind(\tdplotCsAlp)*sind(\tdplotCsBet)},%
        \fpeval{\tdplotCsZe*cosd(\tdplotCsBet)}}%
      \tikzset{3d/visible/.style={tdplotCsFront},3d/hidden/.style={tdplotCsBack}}%
      \tdplotCsIfStyleT{#1}{tdplotCsFill}{%
        \tikzset{3d/circle on sphere/fill/.style={fill,tdplotCsFill}}}%
      \path pic{3d/circle on sphere={R={#2},n={(\tdplotCsN)},P={(\tdplotCsP)}}};
    \end{scope}}}
\NewDocumentCommand{\tdplotCsDrawGreatCircle}{O{}mmm}{\tdplotCsDrawCircle[#1]{#2}{#3}{#4}{0}}
\NewDocumentCommand{\tdplotCsDrawLatCircle}{O{}mm}{\tdplotCsDrawCircle[#1]{#2}{0}{0}{#3}}
\NewDocumentCommand{\tdplotCsDrawLonCircle}{O{}mm}{\tdplotCsDrawCircle[#1]{#2}{#3}{90}{0}}
\NewDocumentCommand{\tdplotCsFrontsidePoint}{}{\textbullet}
\NewDocumentCommand{\tdplotCsBacksidePoint}{}{\(\circ\)}
% \tdplotCsDrawPoint[<style>]{<radius>}{<alpha>}{<beta>}
\NewDocumentCommand{\tdplotCsDrawPoint}{O{}mmm}{%
  \begin{scope}[#1]
    \pgfmathsetmacro\tdplotCsAlp{#3}%
    \pgfmathsetmacro\tdplotCsBet{#4}%
    \pgfmathsetmacro\tdplotCsRad{#2}%
    \edef\tdplotCsN{\fpeval{\tdplotCsRad*cosd(\tdplotCsAlp)*sind(\tdplotCsBet)},%
      \fpeval{\tdplotCsRad*sind(\tdplotCsAlp)*sind(\tdplotCsBet)},%
      \fpeval{\tdplotCsRad*cosd(\tdplotCsBet)}}%
    \path pic{3d/point on sphere={P={(\tdplotCsN)},
      front/.style={tdplotPtFront},back/.style={tdplotPtBack},
      front text=\tdplotCsFrontsidePoint,back text=\tdplotCsBacksidePoint}};
  \end{scope}}
